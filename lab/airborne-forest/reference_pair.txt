func forest_cost(label):
 var times=[]
 var prev=Time.get_ticks_usec()
 for i in 37:
  await get_tree().process_frame
  await RenderingServer.frame_post_draw
  var t=Time.get_ticks_usec()
  if i>3:times.append((t-prev)*.001)
  prev=t
 times.sort()
 print("FOREST COST ",label," median_ms=",times[times.size()/2]," p95=",times[int(times.size()*.95)]," triangles=",RenderingServer.get_rendering_info(RenderingServer.RENDERING_INFO_TOTAL_PRIMITIVES_IN_FRAME))
